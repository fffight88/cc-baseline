// Hybrid search over product catalog
db.products.aggregate([
  { $vectorSearch: {
    index: "catalog_embeddings",
    path: "embedding",
    queryVector: embed("waterproof trail shoes"),
    numCandidates: 200, limit: 5
  }},
  { $match: { inStock: true, price: { $lt: 150 } } },
  { $project: { name: 1, price: 1, score: { $meta: "vectorSearchScore" } } }
])
// → 5 documents · 23 ms
