-- Top customers by revenue this month
select p.username,
       sum(o.total) as revenue,
       count(*) as orders
from orders o
join profiles p on p.id = o.user_id
where o.created_at > date_trunc('month', now())
group by 1 order by revenue desc
limit 10;
